package com.cm6123.monopoly;

import java.util.ArrayList;
import java.util.List;
import java.util.Random;
//random import for stations

public final class Board {

    /**
     * Field that can hold 16 items.
     * Squares is the name of the field.
     */
    private final List<Property> squares;

    /**
     * Field that holds underground stations.
     * undergroundStations is the name of the field.
     */
    private final List<UndergroundStation> undergroundStations;

    /**
     * Random number generator used to pick the station the current player moves to.
     */
    private final Random randomstation = new Random();

    /**
     * Constructs a new board with a given list of squares.
     * @param squareList list of 16 squares null for roads
     * Initialize undergroundStations list as new ArrayList<>.
     * Empty list for stations to be added.
     */
    public Board(final List<Property> squareList) {
        this.squares = squareList;
        undergroundStations = new ArrayList<>();
    }

    /**
     * Returns square at given position.
     * @param position the 0 index
     * @return the property position
     */
    public Object getSquare(final int position) {
        return squares.get(position);
    }

    /**
     * Returns the number of squares on the board.
     * @return the board sixe (16)
     */
    public int getSize() {
        return squares.size();
    }

    /**
     * Adds underground station to the board's list.
     * @param undergroundStation object to add
     */
    public void addUndergroundStation(final UndergroundStation undergroundStation) {
            undergroundStations.add(undergroundStation);
        }

    /**
     * Returns a random station different from the one the player is on.
     * @param currentPosition is excluded
     * @return a random station from the remaining ones
     */
    public int getOtherStation(final int currentPosition) {
        List<Integer> otherpositions = new ArrayList<>();
        for (UndergroundStation us : undergroundStations) {
            int station = us.getStationPosition();
            if (station != currentPosition) {
                otherpositions.add(station);
            }
        }
        int randomindex = randomstation.nextInt(otherpositions.size());
        return otherpositions.get(randomindex);
        //after loop pick random index from the list of remaining positions
    }

    /**
     * Checks if the square the current player landed on is a station.
     * @param position board index to check
     * @return true of the square is a station and false if it's not
     */
    public boolean getUndergroundStation(final int position) {
        for (UndergroundStation us : undergroundStations) {
            if (us.getStationPosition() == position) {
                return true;
            }
        }
        return false;
    }

    /**
     * Clears all underground stations from the board.
     * This method is primarily used for testing purposes to ensure
     * a clean state before adding specific stations for test scenarios.
     */
    public void clearUndergroundStations() {
        undergroundStations.clear();
    }
}

    public void playRounds(final int numberOfRounds, final Scanner scanner) {

        for (int round = 0; round < numberOfRounds; round++) {
            System.out.println("Round " + (round + 1) + " begins!");
            //to see if it is actually doing rounds

            for (Player currentplayer : players) {
                handlePlayerTurn(currentplayer, scanner);

     * Provides movement logic to rollAndMove() method
     *
     * @param currentplayer the player whose turn it is
     * @param scanner       the Scanner for user input
     */
    private void handlePlayerTurn(final Player currentplayer, final Scanner scanner) {
        if (cycleManager.getCycleRacePosition(currentplayer.getPosition())) {
            //Check if the player is on a cycle race square at the start of their turn
            currentplayer.setCycleRestriction(true);
        }
        if (currentplayer.getBalance() < 0) {
            // Check bankruptcy at start of turn
            bankruptcyManager.handleBankruptcy(currentplayer);
        }
        if (!players.contains(currentplayer)) {
            //skip turn if player was removed
            return;
            //CHANGED from continue to return
        }

        //STATION CHECK — before the dice roll
        if (board.getUndergroundStation(currentplayer.getPosition()) && !currentplayer.getJustTeleported()) {
            System.out.println("Station detected — teleporting!");
            currentplayer.setJustTeleported(true);
            playerOnSquare(currentplayer, scanner);
            System.out.println("Continue reached — skipping dice roll!");
            return;
            //CHANGED from continue to return
        }
        rollAndMove(currentplayer, scanner);
        // Roll dice and move
    }

    /**
     * Rolls the dice, applies the cycle restrictions, moves the player and handles passing home.
     * @param currentplayer the player whose tuurn it is
     * @param scanner the scanner for user input
     */
    private void rollAndMove(final Player currentplayer, final Scanner scanner) {
        int oldPosition = currentplayer.getPosition();
        int steps = movementManager.calculateSteps(currentplayer, cycleManager.getCycleRacePositions());
        currentplayer.move(steps, board.getSize());

        System.out.println("Player " + currentplayer + " rolled " + steps
                + " and moved to square " + currentplayer.getPosition()
                + " Balance: " + currentplayer.getBalance());

        if (currentplayer.getPosition() < oldPosition) {
            movementManager.addPassingHomeBonus(currentplayer, oldPosition);
        }
        playerOnSquare(currentplayer, scanner);
    }

    /**
     * If only one player remains, they are the winner.
     * Otherwise, the player with the highest assets wins.
     */
    private void declareWinner() {
        if (players.size() == 1) {
            System.out.println("Congratulations " + players.get(0) + "\n"
                    + "with total assets of £" + calculateAssets(players.get(0)) + "!");
        } else {
            Player winner = getWinner();
            System.out.println("Congratulations " + winner + "\n"
                    + "with total assets of £" + calculateAssets(winner) + "!");
        }
    }

    /**
     * Calculate all the assets of the player.
     * Assests = balance + total purchase price of all owned properties.
     * @param player the thayer whose assets are to be calculated.
     * @return the total asset value of the player.
     */
    public int calculateAssets(final Player player) {
        //Application.java needs to display the winner's assets.
        // Since calculateAssets() is a method that calculates player assets,
        // it makes sense to be public so other classes can use it.
        int totalPropertyValue = 0;

        for (int i = 0; i < board.getSize(); i++) {
            Object square = board.getSquare(i);

            if (square instanceof Property property && property.getPropertyOwner() == player) {
                //if the square is a property and the player owns it
                totalPropertyValue = totalPropertyValue + property.getPropertyPurchasePrice();
            }
        }
        return player.getBalance() + totalPropertyValue;
    }

    /**
     * Returns the player with the highest total assets.
     * Assets = cash balance + total purchase price of all owned properties.
     * @return the player with the highest assets
     */
    public Player getWinner() {
        Player winner = players.get(0);
        int playerWithHighestAssets = calculateAssets(winner);

        for (Player consideredPlayer : players) {
            // Loop through each player to find the one with the highest asset
            // Calculate assets for the current player being checked

            int playerAssets = calculateAssets(consideredPlayer);

            if (playerAssets > playerWithHighestAssets) {
                playerWithHighestAssets = playerAssets;
                winner = consideredPlayer;
            }
        }
        return winner;
    }

    /**
     * What to do with the square the player is standing on.
     * @param currentplayer the player whose turn it currently is
     * @param scanner the scanner used for user input
     */
    public void playerOnSquare(final Player currentplayer, final Scanner scanner) {
        boolean movingForward = false;
        //declare movingForward variable
        //player's current pos to check for us for pass HOME check
        int currentPosition = currentplayer.getPosition();
        if (board.getUndergroundStation(currentPosition)) {
            //if statement calls method in Board.java to check square status
            int newPosition = board.getOtherStation(currentPosition);
            //return RANDOM pos for another station
            //store it as var newPosition so player will move
            currentplayer.setPosition(newPosition);
            //player moved to a different station imediately
            if (newPosition < currentPosition) {
                movingForward = true;
                //pass rules for Home
                //direction matters to see if player recieves money
                int reward = movementManager.getReward(currentplayer);
                currentplayer.setBalance(currentplayer.getBalance() + reward);
            }
            return;
        }
        if (cycleManager.getCycleRacePosition(currentPosition)) {
            currentplayer.setCycleRestriction(true);
            return;
        }
        //Handle property or ther squares
        propertyManager.handleNonStationSquare(currentplayer, scanner);
    }

        /**

         * @param player the player to check
         */

    }
}

     * @param boardSize the total number of squares on the board
     * @param random    the Random object used the generate the price
     * @return te generated purchase price

    private static int generatePurchasePrice(final int i, final int boardSize, final Random random) {
        //helper method so priivate
        int purchasePrice = 0;
        if (i < boardSize / 3) {
            //If the player lands on the first third of the board
            purchasePrice = 10 + random.nextInt(191);
            //The price will be between 10 and 200
        } else if (i < 2 * (boardSize / 3)) {

            purchasePrice = 200 + random.nextInt(111);

        } else {
            //Last third of the board do end with else
            purchasePrice = 310 + random.nextInt(191);

    private static Property createSquare(final int i, final int boardSize, final Random random, final List<String> propertyList, final List<Integer> usedPriceValues) {

  private Queue<Integer> forcedRolls = new LinkedList<>();

      int result = forcedRolls.poll();
      //removes 1st num and returns it
      System.out.println("Forced roll used: " + result);
      //To confirm forced roll is being applied
      return result;
    }
    Double tempRoll = Math.ceil(Math.random() * faces.doubleValue());
    //if the queue is empty the code runs
    //generates random number between 1 and 6(faces)
    return tempRoll.intValue();
  }

  /**
   * Clears the forced roll value.
   * Allows the dice to return to ramdom rolls
   */
  public void clearForcedRoll() {
    forcedRolls.clear();
    //removes all values from the queue
  }

  public void setNextRoll(final Integer roll) {

        Property cheapest = null;
